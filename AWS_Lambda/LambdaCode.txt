#Change dynamodb name and SNS Topic ARN 

import json
import boto3

# Initialize DynamoDB
dynamodb = boto3.resource("dynamodb")
table = dynamodb.Table("ResumeFiles")

# Initialize SNS
sns = boto3.client("sns")

# SNS Topic ARN
TOPIC_ARN = "arn:aws:sns:ap-south-1:206388234317:ResumeNotification"

def lambda_handler(event, context):

    # Get S3 event details
    record = event["Records"][0]

    bucket = record["s3"]["bucket"]["name"]
    file_name = record["s3"]["object"]["key"]
    file_size = record["s3"]["object"]["size"]
    event_time = record["eventTime"]

    # Store file details in DynamoDB
    table.put_item(
        Item={
            "FileName": file_name,
            "Bucket": bucket,
            "FileSize": file_size,
            "UploadTime": event_time
        }
    )

    # Publish notification to SNS
    sns.publish(
        TopicArn=TOPIC_ARN,
        Subject="📄 New Resume Uploaded",
        Message=f"""
A new resume has been uploaded successfully.

File Name   : {file_name}
Bucket Name : {bucket}
File Size   : {file_size} bytes
Upload Time : {event_time}

Regards,
AWS Lambda
"""
    )

    print(f"File '{file_name}' stored successfully in DynamoDB.")
    print("SNS notification sent successfully.")

    return {
        "statusCode": 200,
        "body": json.dumps("Resume processed successfully.")
    }
